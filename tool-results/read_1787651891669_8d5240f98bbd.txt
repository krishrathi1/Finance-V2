   169→function sentimentDot(sentiment: number): string {
   170→  if (sentiment > 0.65) return "bg-success";
   171→  if (sentiment < 0.4) return "bg-danger";
   172→  return "bg-warn";
   173→}

   175→function heatTileClass(pct: number): string {
   176→  if (pct > 5) return "bg-success";
   177→  if (pct > 2) return "bg-success/85";
   178→  if (pct > 0.4) return "bg-success/65";
   179→  if (pct > -0.4) return "border border-border/60 bg-panel";
   180→  if (pct > -2) return "bg-danger/60";
   181→  if (pct > -5) return "bg-danger/80";
   182→  return "bg-danger";
   183→}

   719→function HeatmapSection() {
   720→  const [indexKey, setIndexKey] = useState("NIFTY50");
   721→  const [data, setData] = useState<HeatmapPayload | null>(null);
   722→  const openStock = useApp((s) => s.openStock);
   723→
   724→  // Fetcher keyed explicitly so tab clicks can fetch the new index directly
   725→  // (event handlers) while polling keeps hitting the latest selection via ref.
   726→  const loadRef = useRef<() => void>(() => {});
   727→  const selectedRef = useRef(indexKey);
   728→
   729→  const loadFor = useCallback(async (key: string) => {
   730→    try {
   731→      const res = await apiGet<HeatmapPayload>(`/api/market/heatmap?index=${key}`);
   732→      if (selectedRef.current === key) setData(res);
   733→    } catch {
   734→      // keep last snapshot
   735→    }
   736→  }, []);
   737→
   738→  const load = useCallback(() => loadFor(indexKey), [indexKey, loadFor]);
   739→
   740→  useEffect(() => {
   741→    loadRef.current = load;
   742→    selectedRef.current = indexKey;
   743→  }, [load, indexKey]);
   744→
   745→  usePolling(() => {
   746→    loadRef.current();
   747→  }, 30000);
   748→
   749→  const selectIndex = (next: string) => {
   750→    setIndexKey(next);
   751→    selectedRef.current = next;
   752→    loadFor(next);
   753→  };
   754→
   755→  return (
   756→    <section aria-label="Index heatmap">
   757→      <SectionHeading
   758→        icon={BarChart3}
   759→        kicker="Index Heatmap"
   760→        title="Live Sector Heatmap"
   761→        right={
   762→          <div className="flex flex-wrap justify-end gap-1.5">
   763→            {INDEX_TABS.map((tab) => (
   764→              <button
   765→                key={tab.value}
   766→                type="button"
   767→                onClick={() => selectIndex(tab.value)}
   768→                aria-pressed={indexKey === tab.value}
   769→                className={cn(
   770→                  "min-h-9 rounded-full border px-3 py-1.5 text-[11px] font-semibold transition",
   771→                  indexKey === tab.value
   772→                    ? "border-brand/40 bg-brand/15 text-brand"
   773→                    : "border-border/60 bg-panel/60 text-muted-foreground hover:border-brand/30 hover:text-text"
   774→                )}
   775→              >
   776→                {tab.label}
   777→              </button>
   778→            ))}
   779→          </div>
   780→        }
   781→      />
   782→
   783→      <div className="rounded-[28px] border border-border/50 bg-panel/50 p-4 backdrop-blur-sm sm:p-5">
   784→        {/* legend + index quote */}
   785→        <div className="flex flex-wrap items-center justify-between gap-2">
   786→          <div className="flex flex-wrap items-center gap-2.5" aria-label="Heatmap color legend">
   787→            {HEAT_LEGEND.map((step) => (
   788→              <span key={step.label} className="flex items-center gap-1">
   789→                <span className={cn("h-2.5 w-2.5 rounded-[3px]", step.swatch)} aria-hidden="true" />
   790→                <span className="text-[10px] tabular-nums text-muted-foreground">{step.label}</span>
   791→              </span>
   792→            ))}
   793→          </div>
   794→          {data?.index && (
   795→            <span className="flex items-center gap-2">
   796→              <span className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   797→                {data.indexName ?? data.index.name}
   798→              </span>
   799→              <span className="font-display text-sm font-bold tabular-nums text-text">
   800→                {data.index.price.toLocaleString("en-IN")}
   801→              </span>
   802→              <ChangePill value={data.index.changePercent} size="xs" />
   803→            </span>
   804→          )}
   805→        </div>
   806→
   807→        {/* tiles */}
   808→        {!data ? (
   809→          <div className="mt-4 grid gap-1.5 [grid-template-columns:repeat(auto-fill,minmax(96px,1fr))]">
   810→            {Array.from({ length: 16 }).map((_, i) => (
   811→              <div
   812→                key={i}
   813→                className="h-[62px] rounded-xl shimmer"
   814→                aria-hidden="true"
   815→              />
   816→            ))}
   817→          </div>
   818→        ) : (
   819→          <div className="mt-4 grid gap-1.5 [grid-template-columns:repeat(auto-fill,minmax(96px,1fr))]">
   820→            {data.rows.map((row) => {
   821→              const neutral = row.changePercent >= -0.4 && row.changePercent <= 0.4;
   822→              return (
   823→                <button
   824→                  key={row.symbol}
   825→                  type="button"
   826→                  onClick={() => openStock(row.symbol)}
   827→                  aria-label={`${row.name} ${fmtPct(row.changePercent)}`}
   828→                  className={cn(
   829→                    "rounded-xl p-2.5 text-left transition hover:scale-[1.03] hover:ring-2 hover:ring-white/40",
   830→                    heatTileClass(row.changePercent),
   831→                    neutral ? "text-text" : "text-white"
