    56→// ── Color coding helpers ───────────────────────────────────
    57→function peColor(pe: number | null | undefined): string {
    58→  if (pe === null || pe === undefined) return "text-muted";
    59→  if (pe <= 0) return "text-muted";
    60→  if (pe < 15) return "text-success font-semibold"; // cheap
    61→  if (pe < 30) return "text-text"; // fair
    62→  if (pe < 50) return "text-amber-400"; // slightly expensive
    63→  return "text-danger"; // very expensive
    64→}
    65→function divColor(div: number | null | undefined): string {
    66→  if (div === null || div === undefined) return "text-muted";
    67→  if (div >= 3) return "text-success font-semibold";
    68→  if (div >= 1) return "text-text";
    69→  return "text-muted";
    70→}

    72→// ── Match-reason helper ────────────────────────────────────
    73→function getMatchReasons(stock: ScreenerResult, filters: ScreenerFilters): string[] {
    74→  const reasons: string[] = [];
    75→  if (filters.sector && stock.sector) reasons.push(`Sector: ${stock.sector}`);
    76→  if (filters.market_cap_min > 0 || filters.market_cap_max > 0) {
    77→    reasons.push(`Market Cap filter matched`);
    78→  }
    79→  if (filters.pe_min > 0 || filters.pe_max > 0) {
    80→    if (stock.pe != null) reasons.push(`PE ${stock.pe.toFixed(1)} within range`);
    81→  }
    82→  if (filters.dividend_min > 0) {
    83→    if (stock.dividendYield != null) reasons.push(`Div Yield ${stock.dividendYield.toFixed(1)}% ≥ ${filters.dividend_min}%`);
    84→  }
    85→  if (reasons.length === 0) reasons.push(`Matched ${filters.exchange || "NSE"} exchange filter`);
    86→  return reasons;
    87→}

   153→/**
   154→ * `crore` arrives already denominated in rupees crore — the screener universe
   155→ * converts it (`marketCap: q.marketCap / 1e7`) before the row ever reaches the
   156→ * client. This used to re-apply a USD->INR rate and a second /1e7 left over
   157→ * from when the rows came from FMP in dollars, which rendered HDFC Bank's
   158→ * ₹11.6 lakh crore as "~₹10 Cr" — wrong by five orders of magnitude, and
   159→ * plausible enough on screen to pass unnoticed.
   160→ */
   161→function formatMarketCap(crore: number): string {
   162→  if (!crore) return "-";
   163→  if (crore >= 100_000) return `₹${(crore / 100_000).toFixed(2)}L Cr`;
   164→  if (crore >= 1_000) return `₹${(crore / 1_000).toFixed(1)}K Cr`;
   165→  return `₹${Math.round(crore)} Cr`;
   166→}
   167→
   168→function formatNumber(value: number | null | undefined, decimals = 2): string {
   169→  if (value === null || value === undefined || isNaN(value)) return "-";
   170→  return value.toFixed(decimals);
   171→}

   173→function buildStockHref(symbol: string, exchange?: string) {
   174→  const normalizedExchange = String(exchange || "NSE").trim().toUpperCase() || "NSE";
   175→  return normalizedExchange === "NSE" ? `/stocks/${symbol}` : `/stocks/${symbol}?exchange=${normalizedExchange}`;
   176→}

   224→  const [aiMode, setAiMode] = useState(false);
   225→  const aiInputRef = useRef<HTMLInputElement>(null);
   226→  // Shared across applyFilters and handleAISearch — both write the same
   227→  // `results`/`aiMode` state, so a slower request from either one must not
   228→  // overwrite a faster, more recent request from either.
   229→  const resultsRequestId = useRef(0);
   230→
   231→  const applyFilters = useCallback(async () => {
   232→    const requestId = ++resultsRequestId.current;
   233→    setLoading(true);
   234→    setError(null);
   235→    setHasSearched(true);
   236→    setAiMode(false);
   237→    setAiParsedFilters(null);
   238→    try {
   239→      const data = await fetchScreenerResults(filters);
   240→      if (requestId !== resultsRequestId.current) return;
   241→      setResults(data);
   242→    } catch (err) {
   243→      if (requestId !== resultsRequestId.current) return;
   244→      setError(err instanceof Error ? err.message : "Failed to fetch results");
   245→      setResults([]);
   246→    } finally {
   247→      if (requestId === resultsRequestId.current) setLoading(false);
   248→    }
   249→  }, [filters]);
   250→
   251→  const handleAISearch = useCallback(
   252→    async (query?: string) => {
   253→      const q = (query ?? aiQuery).trim();
   254→      if (!q) return;
   255→      const requestId = ++resultsRequestId.current;
   256→      setAiLoading(true);
   257→      setAiError(null);
   258→      setAiMode(true);
   259→      setHasSearched(true);
   260→      try {
   261→        const { results: aiResults, parsedFilters } = await fetchAIScreenerResults(q);
   262→        if (requestId !== resultsRequestId.current) return;
   263→        setResults(aiResults);
   264→        setAiParsedFilters(parsedFilters);
   265→      } catch {
   266→        if (requestId !== resultsRequestId.current) return;
   267→        setAiError("AI screener unavailable. Try using manual filters.");
   268→        setAiMode(false);
   269→      } finally {
   270→        if (requestId === resultsRequestId.current) setAiLoading(false);
   271→      }
   272→    },
   273→    [aiQuery]
   274→  );
   275→
   276→  const resetFilters = () => {
   277→    setFilters({ ...DEFAULT_FILTERS });
   278→    setActiveCapPreset(null);
   279→    setResults([]);
   280→    setHasSearched(false);
   281→    setAiMode(false);
   282→    setAiQuery("");
   283→    setAiParsedFilters(null);
   284→    setError(null);
   285→  };
   286→
   287→  const handleSort = (key: SortKey) => {
   288→    if (sortKey === key) {
   289→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   290→    } else {
   291→      setSortKey(key);
   292→      setSortDir(key === "symbol" || key === "companyName" || key === "sector" ? "asc" : "desc");
   293→    }
   294→  };
   295→
   296→  const sortedResults = [...results].sort((a, b) => {
   297→    const aVal = a[sortKey];
   298→    const bVal = b[sortKey];
   299→    if (typeof aVal === "string" && typeof bVal === "string") {
   300→      return sortDir === "asc" ? aVal.localeCompare(bVal) : bVal.localeCompare(aVal);
   301→    }
   302→    const aNum = typeof aVal === "number" ? aVal : 0;
   303→    const bNum = typeof bVal === "number" ? bVal : 0;
   304→    return sortDir === "asc" ? aNum - bNum : bNum - aNum;
   305→  });

   307→  const updateFilter = <K extends keyof ScreenerFilters>(key: K, value: ScreenerFilters[K]) => {
