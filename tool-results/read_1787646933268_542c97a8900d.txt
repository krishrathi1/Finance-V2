    37→async function fetchWithTimeout(url: string, init: RequestInit, timeoutMs: number) {
    38→  const controller = new AbortController();
    39→  const timeout = setTimeout(() => controller.abort(), timeoutMs);
    40→  try {
    41→    return await fetch(url, { ...init, signal: controller.signal });
    42→  } finally {
    43→    clearTimeout(timeout);
    44→  }
    45→}

    47→function getFreshCache<T>(key: string, maxAgeMs: number): T | null {
    48→  const hit = memoryCache.get(key);
    49→  if (!hit) return null;
    50→  if (Date.now() - hit.at > maxAgeMs) return null;
    51→  return hit.data as T;
    52→}
    53→
    54→function getStaleCache<T>(key: string): T | null {
    55→  const hit = memoryCache.get(key);
    56→  if (!hit || Date.now() - hit.at > MAX_STALE_AGE_MS) {
    57→    if (hit) memoryCache.delete(key);
    58→    return null;
    59→  }
    60→  return hit.data as T;
    61→}
    62→
    63→function setCache<T>(key: string, data: T) {
    64→  const now = Date.now();
    65→  for (const [cacheKey, entry] of memoryCache) {
    66→    if (now - entry.at > MAX_STALE_AGE_MS) memoryCache.delete(cacheKey);
    67→  }
    68→  if (!memoryCache.has(key) && memoryCache.size >= MAX_MEMORY_CACHE_ENTRIES) {
    69→    const oldestKey = memoryCache.keys().next().value as string | undefined;
    70→    if (oldestKey) memoryCache.delete(oldestKey);
    71→  }
    72→  memoryCache.set(key, { at: now, data });
    73→}
    74→
    75→const inFlight = new Map<string, Promise<unknown>>();
    76→
    77→/**
    78→ * Coalesce concurrent cache-miss calls for the same key into one shared
    79→ * request. Without this, N simultaneous requests for the same resource (e.g.
    80→ * the dashboard for RELIANCE loaded from several components at once) each
    81→ * independently miss the cache and fire their own upstream fetch.
    82→ */
    83→function dedupeInFlight<T>(key: string, fn: () => Promise<T>): Promise<T> {
    84→  const pending = inFlight.get(key);
    85→  if (pending) return pending as Promise<T>;
    86→  const promise = fn().finally(() => {
    87→    inFlight.delete(key);
    88→  });
    89→  inFlight.set(key, promise);
    90→  return promise;
    91→}
    92→
    93→function isAbortError(error: unknown) {
    94→  if (!error) return false;
    95→  if (error instanceof Error && error.name === "AbortError") return true;
    96→  return String(error).toLowerCase().includes("aborted");
    97→}

    99→export async function fetchDashboardEnvelope(
   100→  symbol: string,
   101→  options: { requestOrigin?: string; exchange?: string; force?: boolean } = {}
   102→): Promise<DashboardEnvelope> {
   103→  const normalizedExchange = String(options.exchange || "NSE").trim().toUpperCase() || "NSE";
   104→  const force = Boolean(options.force);
   105→  const key = `dashboard:${symbol.toUpperCase()}:5Y:${normalizedExchange}`;
   106→  const fresh = force ? null : getFreshCache<DashboardEnvelope>(key, 30_000);
   107→  if (fresh) return fresh;
   108→
   109→  const stale = getStaleCache<DashboardEnvelope>(key);
   110→
   111→  return dedupeInFlight(key, async () => {
   112→    const attempts = force
   113→      ? [{ timeoutMs: 20_000, refresh: true }]
   114→      : [
   115→          { timeoutMs: 15_000, refresh: false },
   116→          { timeoutMs: 20_000, refresh: true }
   117→        ];
   118→    let lastError: unknown = null;
   119→
   120→    for (const attempt of attempts) {
   121→      try {
   122→        const res = await fetchWithTimeout(
   123→          getApiUrl(`/${symbol}/dashboard?timeframe=5Y&exchange=${encodeURIComponent(normalizedExchange)}${attempt.refresh ? "&refresh=true" : ""}`),
   124→          {
   125→            cache: "no-store"
   126→          },
   127→          attempt.timeoutMs
   128→        );
   129→        if (!res.ok) {
   130→          throw new Error(`Dashboard request failed: ${res.status}`);
   131→        }
   132→        const payload = await res.json();
   133→        const envelope = {
   134→          data: payload.data as DashboardData,
   135→          cached: Boolean(payload.cached),
   136→          stale: Boolean(payload.stale),
   137→          fallback: Boolean(payload.fallback),
   138→          warning: typeof payload.warning === "string" ? payload.warning : undefined
   139→        } satisfies DashboardEnvelope;
   140→        setCache(key, envelope);
   141→        return envelope;
   142→      } catch (error) {
   143→        lastError = error;
   144→      }
   145→    }
   146→
   147→    if (stale) return { ...stale, cached: true, stale: true };
   148→    if (isAbortError(lastError)) {
   149→      throw new Error("Dashboard request timed out. Please retry in a few seconds.");
   150→    }
   151→    throw (lastError instanceof Error ? lastError : new Error("Dashboard request failed"));
   152→  });
   153→}

   302→/**
   303→ * The ticker endpoint bounds each request to 50 symbols so one call can't fan
   304→ * out to hundreds of upstream quotes. That bound is deliberate and stays — but
   305→ * it is a *per-request* limit, not a limit on what a user may hold. Sending a
   306→ * longer list in one request had the server silently keep the first 50 and
   307→ * discard the rest, so a portfolio of 60 stocks priced 50 and left the other
   308→ * 10 blank while the total quietly understated itself. Callers pass whole
   309→ * portfolios (up to 500 holdings), watchlists and alert lists, so the split
   310→ * belongs here.
   311→ */
   312→export const TICKER_SYMBOLS_PER_REQUEST = 50;
   313→
   314→export async function fetchTickerTape(symbols: string[] = [], options: { force?: boolean } = {}) {
   315→  type TickerTapeRow = { symbol: string; cmp: number; change: number; changePercent: number; exchange?: "NSE" | "BSE" };
   316→  const force = Boolean(options.force);
   317→  const key = `ticker:${symbols.join(",") || "default"}`;
   318→  const fresh = force
   319→    ? null
   320→    : getFreshCache<TickerTapeRow[]>(key, 10_000);
   321→  if (fresh) return fresh;
   322→
   323→  const stale = getStaleCache<TickerTapeRow[]>(key);
   324→
   325→  const batches: string[][] = [];
   326→  for (let index = 0; index < symbols.length; index += TICKER_SYMBOLS_PER_REQUEST) {
   327→    batches.push(symbols.slice(index, index + TICKER_SYMBOLS_PER_REQUEST));
   328→  }
   329→  // No symbols means "the whole market snapshot", which is a single call with
   330→  // its own much longer budget — not a batch of zero.
   331→  if (batches.length === 0) batches.push([]);
   332→
   333→  const requestUrl = (batch: string[]) => {
   334→    const queryParts: string[] = [];
   335→    if (batch.length) queryParts.push(`symbols=${encodeURIComponent(batch.join(","))}`);
   336→    if (force) queryParts.push("refresh=true");
   337→    return getApiUrl(`/ticker${queryParts.length ? `?${queryParts.join("&")}` : ""}`);
   338→  };
   339→
   340→  return dedupeInFlight(key, async () => {
   341→    try {
   342→      const timeoutMs = symbols.length ? (force ? 7000 : 4500) : (force ? 35_000 : 25_000);
   343→
   344→      const results = await Promise.allSettled(
   345→        batches.map(async (batch) => {
   346→          const res = await fetchWithTimeout(requestUrl(batch), { cache: "no-store" }, timeoutMs);
   347→          if (!res.ok) throw new Error(`Ticker request failed: ${res.status}`);
   348→          const payload = await res.json();
   349→          return (payload.data || []) as TickerTapeRow[];
   350→        })
   351→      );
   352→
   353→      // A batch that fails costs its own symbols their price, not everyone
   354→      // else's — partial pricing beats a blank portfolio. Only a total failure
   355→      // falls through to the stale copy.
   356→      const succeeded = results.filter(
   357→        (result): result is PromiseFulfilledResult<TickerTapeRow[]> => result.status === "fulfilled"
   358→      );
   359→      if (!succeeded.length) {
   360→        throw results[0]?.status === "rejected" ? results[0].reason : new Error("Ticker request failed");
   361→      }
   362→
   363→      const rows = succeeded.flatMap((result) => result.value);
   364→      if (!rows.length && stale?.length) return stale;
   365→      // Only cache a complete answer. Caching a partial one would pin the
   366→      // missing symbols blank for the whole TTL.
   367→      if (succeeded.length === batches.length) setCache(key, rows);
   368→      return rows;
   369→    } catch (err) {
   370→      if (stale) return stale;
   371→      throw err;
   372→    }
   373→  });
   374→}

   427→export async function fetchIndexHeatmap(indexName: string, options: { force?: boolean } = {}) {
   428→  const force = Boolean(options.force);
   429→  const query = `?index=${encodeURIComponent(indexName)}${force ? "&refresh=true" : ""}`;
   430→  const key = `heatmap:${indexName.toUpperCase()}`;
   431→  const fresh = force
   432→    ? null
   433→    : getFreshCache<IndexHeatmapPayload>(key, 15_000);
   434→  if (fresh) return fresh;
   435→
   436→  const stale = getStaleCache<IndexHeatmapPayload>(key);
   437→
   438→  return dedupeInFlight(key, async () => {
   439→    try {
   440→      const res = await fetchWithTimeout(getApiUrl(`/index-heatmap${query}`), { cache: "no-store" }, force ? 9000 : 6000);
   441→      if (!res.ok) {
   442→        throw new Error(`Index heatmap request failed: ${res.status}`);
   443→      }
   444→      const payload = await res.json();
   445→      const data: IndexHeatmapPayload = {
   446→        indexName: (payload.indexName || indexName) as string,
   447→        updatedAt: (payload.updatedAt || "") as string,
   448→        rows: (payload.rows || []) as Array<{ symbol: string; cmp: number; change: number; changePercent: number; exchange?: "NSE" | "BSE" }>,
   449→        source: typeof payload.source === "string" ? payload.source : "",
   450→        constituentCount: typeof payload.constituentCount === "number" ? payload.constituentCount : 0,
   451→      };
   452→
   453→      // A 200 response with no rows means every upstream provider failed this
   454→      // call (soft failure, not an HTTP error) — prefer showing the last good
   455→      // data over blanking the heatmap, same as the network-error path below.
   456→      if (!data.rows.length && stale?.rows.length) {
   457→        return stale;
   458→      }
   459→
   460→      setCache(key, data);
   461→      return data;
   462→    } catch (err) {
   463→      if (stale) return stale;
   464→      throw err;
   465→    }
   466→  });
   467→}

   545→
   546→  const stale = getStaleCache<
   547→    Array<{
   548→      title: string;
   549→      source: string;
   550→      publishedAt: string;
   551→      url: string;
   552→      summary: string;
   553→      imageUrl: string | null;
   554→    }>
   555→  >(key);
   556→
   557→  return dedupeInFlight(key, async () => {
   558→    try {
   559→      const query = force ? "?refresh=true" : "";
   560→      const res = await fetchWithTimeout(getApiUrl(`/market-news${query}`), { cache: "no-store" }, force ? 10000 : 7000);
   561→      if (!res.ok) {
   562→        throw new Error(`Market news request failed: ${res.status}`);
   563→      }
   564→      const payload = await res.json();
   565→      const rows = (payload.data || []) as Array<{
   566→        title: string;
   567→        source: string;
   568→        publishedAt: string;
   569→        url: string;
   570→        summary: string;
   571→        imageUrl: string | null;
   572→      }>;
   573→      if (rows.length) {
   574→        setCache(key, rows);
   575→        return rows;
   576→      }
   577→      if (stale) return stale;
   578→      return rows;
   579→    } catch (err) {
   580→      if (stale) return stale;
   581→      throw err;
   582→    }
   583→  });
   584→}

   721→export async function fetchScreenerResults(filters: ScreenerFilters): Promise<ScreenerResult[]> {
   722→  const params = new URLSearchParams();
   723→  if (filters.exchange) params.set("exchange", filters.exchange);
   724→  if (filters.sector) params.set("sector", filters.sector);
   725→  if (filters.industry) params.set("industry", filters.industry);
   726→  if (filters.market_cap_min) params.set("market_cap_min", String(filters.market_cap_min));
   727→  if (filters.market_cap_max) params.set("market_cap_max", String(filters.market_cap_max));
   728→  if (filters.pe_min) params.set("pe_min", String(filters.pe_min));
   729→  if (filters.pe_max) params.set("pe_max", String(filters.pe_max));
   730→  if (filters.price_min) params.set("price_min", String(filters.price_min));
   731→  if (filters.price_max) params.set("price_max", String(filters.price_max));
   732→  if (filters.dividend_min) params.set("dividend_min", String(filters.dividend_min));
   733→  if (filters.volume_min) params.set("volume_min", String(filters.volume_min));
   734→  if (filters.limit) params.set("limit", String(filters.limit));
   735→  if (filters.query && filters.query.trim()) params.set("query", filters.query.trim());
   736→
   737→  const url = getApiUrl(`/screener?${params.toString()}`);
   738→  const res = await fetch(url, { cache: "no-store" });
   739→  if (!res.ok) {
   740→    throw new Error(`Screener request failed: ${res.status}`);
   741→  }
   742→  const payload = await res.json();
   743→  return (payload.results || []) as ScreenerResult[];
   744→}

   754→    body: formData,
   755→    cache: 'no-store',
   756→  });
   757→
